#include "stm32f10x.h"                  // Device header
#include "stdbool.h"

#define БЕГИ_ФОРЕСТ_БЕГИ 1

//Предыдущее состояние кнопки
static uint16_t m_btn_prev_state;
static bool m_bit_flag;

//Позже расскажу что это и зачем
void TIM2_IRQHandler(void);
void TIM7_IRQHandler(void);

void TIM1_BRK_TIM15_IRQHandler(void);
void TIM1_UP_TIM16_IRQHandler(void);
void TIM1_TRG_COM_TIM17_IRQHandler(void);

//Значения прироста
static int16_t m_dv_red;
static int16_t m_dv_green;
static int16_t m_dv_blue;

static bool m_drv1_flag;
static bool m_drv2_flag;
static bool m_drv3_flag;

//Делитель, чтобы управление двигателями не такое быстрое было
static uint8_t drv_ctl_psc = 0;
static uint16_t speed_hold = 0;

static int16_t m_drv_dv;

int main(void)
{
	//Разгон жопы
	
	
	m_bit_flag = 0;
	
	m_drv1_flag = 0;
	m_drv2_flag = 0;
	m_drv3_flag = 0;
	
	drv_ctl_psc = 8;
	
	m_drv_dv = 1;
	
	m_dv_red = 2;
	m_dv_green = 2;
	m_dv_blue = 2;
	
	//Настройка порта
	GPIO_InitTypeDef m_gpioc;
	
	//Врубаем тактирование, чтобы до порта доходило то что от него хотят
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);
	
	//Загружаем настройки по умолчанию
	GPIO_StructInit(&m_gpioc);
	
	//Ногодрыг 8 и 9
	m_gpioc.GPIO_Pin = GPIO_Pin_6 | GPIO_Pin_8 | GPIO_Pin_9 | GPIO_Pin_10 | GPIO_Pin_11 | GPIO_Pin_12;
	//Скорость похуй какая пусть будет 2 мегадрочки в секунду
	m_gpioc.GPIO_Speed = GPIO_Speed_2MHz;
	//Порт на выход
	m_gpioc.GPIO_Mode = GPIO_Mode_Out_PP;
	
	//Применяем настройки
	GPIO_Init(GPIOC, &m_gpioc);
	
	//Такой способ показать что прошивка алё
	uint32_t i;
	GPIOC->BSRR = GPIO_Pin_8;
	for(i = 0; i < 50000; ++i)
	{
		//NOP
	}
	GPIOC->BRR = GPIO_Pin_8;
	for(i = 0; i < 50000; ++i)
	{
		//NOP
	}
		
	//----------------------------------------------------------------------------------------------------------
	//Ну да ладно, хуй с этим. Тут тело посерьёзнее есть
		
	//Настройка шим
	TIM_TimeBaseInitTypeDef m_tim2;
	TIM_OCInitTypeDef m_im2pwm;
	
	RCC_APB1PeriphClockCmd(RCC_APB1ENR_TIM2EN, ENABLE);							//Врубаем продрачивание таймера
	
	TIM_TimeBaseStructInit(&m_tim2);																//Загружаем настройки по умолчанию и меняем их
	m_tim2.TIM_Period = 1000;																				//Здесь будет формировать ШИМ. Это значение сумма периодов вкл и выкл - например для 80% будет 800 вкл + 200 выкл
	m_tim2.TIM_Prescaler = 0;																				//Делитель до 1000000 попугаев - 1 МГц в наносек
	
	TIM_OCStructInit(&m_im2pwm);																		//Загружаем настройки по умолчанию, снова
	m_im2pwm.TIM_OCMode = TIM_OCMode_PWM1;													//ШИМка
	m_im2pwm.TIM_OutputState = TIM_OutputState_Enable;							//Вкл, через это будем дёргать ногу
	m_im2pwm.TIM_Pulse = 500;																				//Похуй, пока 50%
	
	//Применяем настройки
	TIM_TimeBaseInit(TIM2, &m_tim2);																//Запуск таймера
	TIM_OC2Init(TIM2, &m_im2pwm);																		//ШИМ Зелёный канал
	TIM_OC3Init(TIM2, &m_im2pwm);																		//ШИМ Синий канал
	TIM_OC4Init(TIM2, &m_im2pwm);																		//ШИМ Красный канал
	
	//Стоямбола, ещё надо ноги настроить
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);						//Врубаем продрачивание порта A
	
	GPIO_InitTypeDef m_gpioa;
	
	GPIO_StructInit(&m_gpioa);																			//Загружаем настройки, что у нас там было-то...
	m_gpioa.GPIO_Mode = GPIO_Mode_AF_PP;														//Переключаем на управление от таймеров
	m_gpioa.GPIO_Pin = GPIO_Pin_1 | GPIO_Pin_2 | GPIO_Pin_3;				//Зелёный, синий и красный цвета
	m_gpioa.GPIO_Speed = GPIO_Speed_2MHz;														//Высокой частоты здесь не будет, так что можно ставить самое лайтовое
	
	//Ладно, можно применять
	GPIO_Init(GPIOA, &m_gpioa);
	
	//Теперь для пущих приколов можно подрубить кнопку
	m_btn_prev_state = 0;
	
	//Это будет кнопочка
	m_gpioa.GPIO_Mode = GPIO_Mode_IPD;															//Подтяжка к питанию, нажатие синей кнопочки заставит заземлить вход
	m_gpioa.GPIO_Pin = GPIO_Pin_0;																	//Синяя кнопка сидит на 0 ноге порта А
	m_gpioa.GPIO_Speed = GPIO_Speed_2MHz;														//Полный рассос
	
	//Применяем настройки
	GPIO_Init(GPIOA, &m_gpioa);
	
	//Генерируем прерывание в котором будем управлять скважиной шима
	RCC_APB1PeriphClockCmd(RCC_APB1ENR_TIM7EN, ENABLE);							//Начинаем дрочку таймера
	
	//Настрока сего господина на хороший лад
	TIM_TimeBaseInitTypeDef m_tim7;
	
	TIM_TimeBaseStructInit(&m_tim7);																//Загружаем настройки по умолчанию и меняем их
	m_tim7.TIM_Period = 1000;																				//Делитель попугаев до 500 герц
	m_tim7.TIM_Prescaler = 64;																			//Делитель до 125000 попугаев - 125 КаГацелов в наносек
	//Сохраяем этот цирк обратно откуда достали
	TIM_TimeBaseInit(TIM7, &m_tim7);
	
	//------------------------------------------------------------------------------------------------------------
	//Управление двиглами
	
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_TIM15, ENABLE);
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_TIM16, ENABLE);
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_TIM17, ENABLE);
	
	TIM_TimeBaseInitTypeDef m_tim_drv;
	
	m_tim_drv.TIM_Period = 300;
	m_tim_drv.TIM_Prescaler = 4;
	
	TIM_TimeBaseInit(TIM15, &m_tim_drv);
	TIM_TimeBaseInit(TIM16, &m_tim_drv);
	TIM_TimeBaseInit(TIM17, &m_tim_drv);
	
	//------------------------------------------------------------------------------------------------------------
	
	//Предустановка
	TIM2->CCR2 = 1000;																						//Меняем шим на 100%
	TIM2->CCR3 = 500;																							//Меняем шим на 50%
	TIM2->CCR4 = 0;																								//Меняем шим на 0%
		
	//Запускаемся
	TIM_ITConfig(TIM2, TIM_IT_Update, ENABLE);											//Врубаем вызов прерывания по заполнению
	TIM_Cmd(TIM2, ENABLE);																					//Да будет свет ёпта
	NVIC_EnableIRQ(TIM2_IRQn);																			//Врубаем обработку прерываний
	
	//Это всё ради прерывания
	TIM_ITConfig(TIM7, TIM_IT_Update, ENABLE);
	TIM_Cmd(TIM7, ENABLE);
	NVIC_EnableIRQ(TIM7_IRQn);
	
	//------------------------------------------------------------------------------------------------------------
	//Управление двиглами
	TIM_ITConfig(TIM15, TIM_IT_Update, ENABLE);
	TIM_ITConfig(TIM16, TIM_IT_Update, ENABLE);
	TIM_ITConfig(TIM17, TIM_IT_Update, ENABLE);
	
	TIM_Cmd(TIM15, ENABLE);
	TIM_Cmd(TIM16, ENABLE);
	TIM_Cmd(TIM17, ENABLE);
	
	NVIC_EnableIRQ(TIM1_BRK_TIM15_IRQn);
	NVIC_EnableIRQ(TIM1_UP_TIM16_IRQn);
	NVIC_EnableIRQ(TIM1_TRG_COM_TIM17_IRQn);
	//------------------------------------------------------------------------------------------------------------
	
	while(БЕГИ_ФОРЕСТ_БЕГИ)
	{
		//Нихуя не делаем
		GPIOC->BSRR = GPIO_Pin_9;
		for(i = 0; i < 50000; ++i)
		{
			//NOP
		}
		GPIOC->BSRR = GPIO_Pin_8;
		for(i = 0; i < 50000; ++i)
		{
			//NOP
		}
		GPIOC->BRR = GPIO_Pin_9;
		for(i = 0; i < 50000; ++i)
		{
			//NOP
		}
		GPIOC->BRR = GPIO_Pin_8;
		for(i = 0; i < 50000; ++i)
		{
			//NOP
		}
	}
}
void TIM2_IRQHandler()
{
	//Обрабатываем прерывание
	
	//Читаем кнопочку
	uint16_t button = 0;
	button = GPIO_ReadInputDataBit(GPIOA, GPIO_Pin_0);							//Опрос синей кнопочки
	TIM_ClearITPendingBit(TIM2, TIM_IT_Update);											//Скидываем флаг
	
	if((button == 1) && (m_btn_prev_state == 0))
	{
		m_btn_prev_state = 1;																					//Переворачиваем флаг
	}
	if((button == 0) && (m_btn_prev_state == 1))
	{
		m_btn_prev_state = 0;																					//Переворачиваем флаг
	}
}
void TIM7_IRQHandler()
{
	//Переворачиваем флаг
	m_bit_flag = !m_bit_flag;
	
	//Сбрасываем флаг прерывания а то охуевание будет
	TIM_ClearITPendingBit(TIM7, TIM_IT_Update);
	
	if(m_bit_flag)
	{
		GPIOC->BRR = GPIO_Pin_6;
	}
	else
	{
		GPIOC->BSRR = GPIO_Pin_6;
	}
	
	//Делаем приколы. Проверяем значения регистров, если больше нужного то dv ставим в минус, если меньше то в плюс
	//Эту петрушку проворачиваем со всеми тремя
	if(TIM2->CCR2 >= 1000)
	{
		m_dv_green = -2;
	}
	else if(TIM2->CCR2 <= 0)
	{
		m_dv_green = 2;
	}
	if(TIM2->CCR3 >= 1000)
	{
		m_dv_blue = -2;
	}
	else if(TIM2->CCR3 <= 0)
	{
		m_dv_blue = 2;
	}
	if(TIM2->CCR4 >= 1000)
	{
		m_dv_red = -2;
	}
	else if(TIM2->CCR4 <= 0)
	{
		m_dv_red = 2;
	}
	if(TIM16->ARR >= 400)
	{
		m_drv_dv = 0; //-1
	}
	else if(TIM16->ARR <= 24)
	{
		if(speed_hold < 1600)
		{
			++speed_hold;
			m_drv_dv = 0;
		}
		else
		{
			speed_hold = 0;
			m_drv_dv = 0; //1
		}
	}
	
	//Меняем значения регистров периода
	TIM2->CCR2 += m_dv_green;	
	TIM2->CCR3 += m_dv_blue;
	TIM2->CCR4 += m_dv_red;
	
	if(drv_ctl_psc > 0)
	{
		--drv_ctl_psc;
	}
	else
	{
		//Управляем двигателями
		TIM15->ARR += m_drv_dv;
		TIM16->ARR += m_drv_dv;
		TIM17->ARR += m_drv_dv;
		
		//Сбрасываем счётчики а то охуевание случится
		TIM15->CNT = 0;
		TIM16->CNT = 0;
		TIM17->CNT = 0;
		drv_ctl_psc = 2;
	}
}
void TIM1_BRK_TIM15_IRQHandler()
{
	m_drv1_flag = !m_drv1_flag;
	TIM_ClearITPendingBit(TIM15, TIM_IT_Update);
	
	TIM15->CNT = 0;
	
	if(m_drv1_flag)
	{
		GPIOC->BRR = GPIO_Pin_10;
	}
	else
	{
		GPIOC->BSRR = GPIO_Pin_10;
	}
}
void TIM1_UP_TIM16_IRQHandler()
{
	m_drv2_flag = !m_drv2_flag;
	TIM_ClearITPendingBit(TIM16, TIM_IT_Update);
	
	TIM16->CNT = 0;
	
	if(m_drv2_flag)
	{
		GPIOC->BRR = GPIO_Pin_11;
	}
	else
	{
		GPIOC->BSRR = GPIO_Pin_11;
	}
}
void TIM1_TRG_COM_TIM17_IRQHandler()
{
	m_drv3_flag = !m_drv3_flag;
	TIM_ClearITPendingBit(TIM17, TIM_IT_Update);
	
	TIM17->CNT = 0;
	
	if(m_drv3_flag)
	{
		GPIOC->BRR = GPIO_Pin_12;
	}
	else
	{
		GPIOC->BSRR = GPIO_Pin_12;
	}
}
